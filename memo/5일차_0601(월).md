# MongoDB이미지 --> python 출력 (이미지 불러오기: base64 -> 바이너리 -> 이미지 객체로 변환(decode)하여 출력)
## MongoDB에서 이미지 데이터 가져오기 + 출력
```py
from PIL import Image

document = collection.find_one({"filename": "test-image"}) # MongoDB에서 이미지 데이터 가져오기
if document:
    img_data = document["data"]
    img_binary = base64.b64decode(img_data) # base64 바이너리로 변환
    img = Image.open(BytesIO(img_binary)) # 메모리에서 바로 이미지 객체 생성
    print("MongoDB에서 이미지를 읽어왔습니다.")
    display(img)
else:
    print("test-image 이미지를 찾을 수 없습니다.")
```

# 나이 분포 히스토그램, 전공별 학생 수 막대그래프, 전공별 나이 분포 박스플롯
```py
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from matplotlib import font_manager, rc
from pymongo import MongoClient

# 한글 폰트 설정 (Windows 기준, Mac은 "AppleGothic")
rc("font", family="Malgun Gothic")
plt.rcParams["axes.unicode_minus"] = False  # 마이너스 깨짐 방지

## DB 연결 및 데이터 불러오기
client = MongoClient("mongodb://localhost:27017/")
db = client["testdb"]
collection = db["students"]

# MongoDB --> DataFrame
cursor = collection.find({})
df = pd.DataFrame(list(cursor))
client.close()

## _id 컬럼 제거
if "_id" in df.columns:
    df = df.drop("_id", axis=1)
print(df)

## 나이 분포 히스토그램: 분포 비교에 많이 사용 (봉우리 위치, 중심 차이 비교)

plt.show()
```

# 새창 띄우기 (canvas/tkinter)

```py
import tkinter as tk
from tkinter import ttk
from pymongo import MongoClient
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg
from matplotlib import font_manager, rc

# 한글 폰트 설정 (Windows용)
font_path = "C:/Windows/Fonts/malgun.ttf"
font_name = font_manager.FontProperties(fname=font_path).get_name()
rc('font', family=font_name)

# MongoDB 연결
client = MongoClient("mongodb://localhost:27017/")
db = client["testdb"]
collection = db["students_csv"]

# 데이터 불러오기
data = list(collection.find({}, {"_id": 0}))
df = pd.DataFrame(data)
client.close()

# Tkinter 메인 윈도우
root = tk.Tk()
root.title("학생 점수 대시보드")
root.geometry("900x700")

# === 제목 ===
title_label = tk.Label(root, text="학생 점수 대시보드", font=("Malgun Gothic", 18, "bold"))
title_label.pack(pady=10)

# === Treeview 테이블 ===
frame_table = tk.Frame(root)
frame_table.pack(pady=10)

columns = list(df.columns)
tree = ttk.Treeview(frame_table, columns=columns, show="headings", height=10)

# 컬럼 헤더 설정
for col in columns:
    tree.heading(col, text=col)
    tree.column(col, width=100, anchor="center")

# 데이터 삽입
for _, row in df.iterrows():
    tree.insert("", "end", values=list(row))
tree.pack()

# === Matplotlib 그래프 표시 ===
frame_graph = tk.Frame(root)
frame_graph.pack(pady=20)

# 과목별 평균 계산
mean_scores = df[["kor", "eng", "math", "science"]].mean()

fig, ax = plt.subplots(figsize=(6, 4))
ax.bar(mean_scores.index, mean_scores.values, color="skyblue")
ax.set_title("과목별 평균 점수")
ax.set_ylabel("점수")
ax.set_ylim(0, 100)

# Tkinter에 matplotlib 그래프 추가
canvas = FigureCanvasTkAgg(fig, master=frame_graph)
canvas.draw()
canvas.get_tk_widget().pack()

# === 종료 버튼 ===
exit_button = tk.Button(root, text="닫기", command=root.destroy, font=("Malgun Gothic", 12))
exit_button.pack(pady=10)

# Tkinter 실행
root.mainloop()
```

## 번외 - 학습/추론 데이터 스케일
- 예측 결과는 운영 데이터인 거죠, 나중에 이 둘 차이가 없는지 학습데이터와 결과데이터 드리프트 검증을 하게 된다.
- 나중에 운영하다 이런 문제가 발생한다.: 트레인 데이터 스케일 프레셔는 100~130, 바이브레이션 0.1~0.3이게 정상. 근데 나중에 템퍼레처: 150, 프레셔 60, 바이브: 0.7 이런식으로 완전다른게 들어옴. 
- 이 스케일 안에서 유형과 패턴 찾아서 예측해야 하는데 신규데이터에서 완전 스케일 다른게 들어오면 기존 학습한 데이터로 예측하면 안맞는다. 이걸 드리프트 감출 해야되는 거다.
- 신규데이터가 스케일 비슷하게 들어오고 있는지.
- 도메인을 파악해야 한다. 담당자가 스케일 조정한경우, 센서고장, 센서 바꿨더니 단위가 달라진 경우, 설계 레시피(공정방법) 달라진 경우 등.. 기존 학습데이터와 스케일이 달라지는 경우 = 허다하다 => ai 예측도 떨어짐 => 드랍하게됨..
- [중요] 해결 방안
    - 스케일 차이나는지 1차적으로 검증, 단위/공정/스케일 달라진 경우 => 신규데이터로 새로학습, 조금 차이나면 머지해서 학습.
    - 드리프트 감지 이후에 -> 모델을 바꿀것인지, 머지해서 학습할 건지, 신규 데이터로 학습할 건지 판단해서 진행해야 한다.
- 학습(머신러닝/딥러닝) -> 실시간 신규 운영서버 데이터 리얼타임으로 상황분석 -> 인퍼런스모델 추론/분석결과 전송 -> 다시 원천디비에 저장 [크게 이런 구조로 돌아가야 한다.]

## 데이터 사이언티스트 / 데이터 엔지니어 구직활동 관련
- 이런 작업 누가?
    - 데이터 사이언티스트: 데이터 분석해서 인사이트 찾아서 의사결정 내리는
    - 데이터 엔지니어: 데이터 분석하고 ~~하고 인프라구축/파이프라인 구축
- 최근 트렌드
    - 분석: 챗지피티가함. 데이터 사이언티스트 희소가치가 떨어짐.
    - 데이터엔지니어 가치가 계속 더 올라가고 있다. 직접 구축해야 하기 때문이다. (환경 정확히 이해해야함 + 그에 맞는 시스템 구축/설계 -> 일일이 설치/테스트/동작테스트 다해야됨==사람이 해야해서 지피티가 하기가 힘든 영역이다.)
    - 지피티는 어디 한곳이 안되어서 솔루션 해결책 듣는 정도는 가능하지만 요즘은 데이터 사이언티스트 줄어들고, 데이터 엔지니어 찾는 회사들 늘어나고 있다. 대신 힘들다. 
    - 땅 파는 작업을 많이 해야한다. 업무를 딱 알고 일하는게 아니라 일단 해보고 테스트를 하는 것.. 그래서 힘들어요. 힘들지만 희소가치는 높다.
    - 데이터사이언티스트 -> 결국엔 도메인 관점 해석이 나와야함 == 도메인에서 활용/효과 낼수있는 해석(인사이트)을 뽑아야됨. 근데 도메인 관련 지식을 데이터사이언티스트가 갖기 쉽지않아서 보통 협업해서 한다. 근데 요즘 지피티가 모델을 잘만듦. -> 도메인 전문가들이 효과낼수있는 방향 생각해서 지피티한테 분석 모델 만들어줘. 이런다.
        - 그래서 데이터사이언티스트 별도로 뽑을일이 없어졌다.
- 도메인 담당자가 데이터 분석 잘하면 진짜 전문가되겠다. 근데 그걸 요즘은 지피티가 채워주고 있다. 요즘은 데이터 사이언티스트만 뽑는기업 별로없다.
- 도메인지식 + 데이터분석이 있으면 경쟁력 있는 것이다.
- [중요] 도메인에 대한 경험 많이 쌓는게 중요하다. -> 빨리 취직해서 도메인업무 경험 쌓는게 먼저 필요하다. 
    - 도메인 업무 -> AI로 효과 낼 수 있는 방안을 계속 생각해야 한다.
- 작은 기업이든 큰 기업이든 도메인 업무 지식 잘 쌓을 수 있는 곳으로 가야한다.
